建表语句

用户表
DROP TABLE IF EXISTS `user`;
CREATE TABLE `user` (
    `id`          INT       NOT NULL AUTO_INCREMENT COMMENT '用户ID',
    `username`    VARCHAR(50)  NOT NULL COMMENT '账号(学号/工号)',
    `password`    VARCHAR(100) NOT NULL COMMENT '密码(加密存储)',
    `real_name`   VARCHAR(50)  DEFAULT NULL COMMENT '真实姓名',
    `phone`       VARCHAR(20)  DEFAULT NULL COMMENT '手机号',
    `car_plate`   VARCHAR(20)  DEFAULT NULL COMMENT '车牌号',
    `role`        INT      NOT NULL DEFAULT 0 COMMENT '角色:0普通用户,1管理员',
    `create_time` DATETIME     DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_username` (`username`)
) ENGINE=InnoDB COMMENT='用户表';

停车区表
DROP TABLE IF EXISTS `parking_area`;
CREATE TABLE `parking_area` (
    `id`          INT       NOT NULL AUTO_INCREMENT COMMENT '区域ID',
    `area_name`   VARCHAR(50)  NOT NULL COMMENT '区域名称,如:教学楼A区',
    `location`    VARCHAR(100) DEFAULT NULL COMMENT '位置描述',
    `description` VARCHAR(200) DEFAULT NULL COMMENT '备注说明',
    PRIMARY KEY (`id`)
) ENGINE=InnoDB COMMENT='停车区表';


车位表
DROP TABLE IF EXISTS `parking_spot`;
CREATE TABLE `parking_spot` (
    `id`          INT      NOT NULL AUTO_INCREMENT COMMENT '车位ID',
    `area_id`     INT      NOT NULL COMMENT '所属停车区ID',
    `spot_code`   VARCHAR(20) NOT NULL COMMENT '车位编号,如:A-001',
    `status`      INT     NOT NULL DEFAULT 0 COMMENT '状态:0空闲,1占用,2已预约,3停用',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_spot_code` (`spot_code`),
    KEY `idx_area_id` (`area_id`)
) ENGINE=InnoDB COMMENT='车位表';


预约表
DROP TABLE IF EXISTS `reservation`;
CREATE TABLE `reservation` (
    `id`          INT      NOT NULL AUTO_INCREMENT COMMENT '预约ID',
    `user_id`     INT      NOT NULL COMMENT '预约用户ID',
    `spot_id`     INT      NOT NULL COMMENT '车位ID',
    `car_plate`   VARCHAR(20) NOT NULL COMMENT '车牌号',
    `start_time`  DATETIME    NOT NULL COMMENT '预约开始时间',
    `end_time`    DATETIME    NOT NULL COMMENT '预约结束时间',
    `status`      INT     NOT NULL DEFAULT 0 COMMENT '状态:0已预约,1使用中,2已完成,3已取消,4超时违约',
    `create_time` DATETIME    DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    KEY `idx_spot_time` (`spot_id`, `start_time`, `end_time`)
) ENGINE=InnoDB COMMENT='预约表';


停车记录表
DROP TABLE IF EXISTS `parking_record`;
CREATE TABLE `parking_record` (
    `id`          INT       NOT NULL AUTO_INCREMENT COMMENT '记录ID',
    `car_plate`   VARCHAR(20)  NOT NULL COMMENT '车牌号',
    `spot_id`     INT       NOT NULL COMMENT '停的车位ID',
    `user_id`     INT       DEFAULT NULL COMMENT '关联用户ID(外来车辆为空)',
    `entry_time`  DATETIME     NOT NULL COMMENT '入场时间',
    `exit_time`   DATETIME     DEFAULT NULL COMMENT '出场时间(为空表示在场中)',
    `fee`         DECIMAL(10,2) DEFAULT 0.00 COMMENT '停车费用',
    `status`      INT      NOT NULL DEFAULT 0 COMMENT '状态:0在场中,1已出场',
    PRIMARY KEY (`id`),
    KEY `idx_car_plate` (`car_plate`),
    KEY `idx_spot_id` (`spot_id`)
) ENGINE=InnoDB COMMENT='停车记录表';


支付表
DROP TABLE IF EXISTS `payment`;
CREATE TABLE `payment` (
    `id`             INT        NOT NULL AUTO_INCREMENT COMMENT '支付ID',
    `record_id`      INT        NOT NULL COMMENT '关联停车记录ID',
    `user_id`        INT        DEFAULT NULL COMMENT '支付用户ID',
    `amount`         DECIMAL(10,2) NOT NULL COMMENT '支付金额',
    `pay_method`     INT       DEFAULT 0 COMMENT '支付方式:0模拟,1支付宝,2微信',
    `pay_status`     INT       NOT NULL DEFAULT 0 COMMENT '状态:0待支付,1已支付,2失败',
    `transaction_no` VARCHAR(50)   DEFAULT NULL COMMENT '交易流水号',
    `pay_time`       DATETIME      DEFAULT NULL COMMENT '支付时间',
    `create_time`    DATETIME      DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`id`),
    KEY `idx_record_id` (`record_id`)
) ENGINE=InnoDB COMMENT='支付表';


违停记录表
DROP TABLE IF EXISTS `violation`;
CREATE TABLE `violation` (
    `id`             INT      NOT NULL AUTO_INCREMENT COMMENT '违停ID',
    `record_id`      INT      DEFAULT NULL COMMENT '关联停车记录ID',
    `car_plate`      VARCHAR(20) NOT NULL COMMENT '车牌号',
    `spot_id`        INT      DEFAULT NULL COMMENT '车位ID',
    `violation_type`  INT     NOT NULL COMMENT '类型:0未预约停车,1超时停车,2乱停乱放',
    `description`    VARCHAR(200) DEFAULT NULL COMMENT '违停描述',
    `handle_status`   INT     NOT NULL DEFAULT 0 COMMENT '处理状态:0未处理,1已处理',
    `create_time`    DATETIME    DEFAULT CURRENT_TIMESTAMP COMMENT '发现时间',
    PRIMARY KEY (`id`)
) ENGINE=InnoDB COMMENT='违停记录表';
